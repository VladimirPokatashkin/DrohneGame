seisu x = { *_* } => x;
seisu y = { *_* } => y;
seisu z = { *_* } => z;

seisu exit1_x = 6;
seisu exit1_y = 1;
seisu exit1_z = 3;

hairetsu visited = { 8, 5, 5 };

shuki i = 0 : 8 kido
    shuki j = 0 : 5 kido
        shuki k = 0 : 5 kido
            visited[i, j, k] = 0;
            k = k + 1;
        shushi
        j = j + 1;
    shushi
    i = i + 1;
shushi

shuki step = 0 : 500 kido


    ronri x1_match = ~(x < exit1_x) ^ ~(x > exit1_x);
    ronri y1_match = ~(y < exit1_y) ^ ~(y > exit1_y);
    ronri z1_match = ~(z < exit1_z) ^ ~(z > exit1_z);
    ronri on_exit1 = x1_match ^ y1_match ^ z1_match;



    sorenara on_exit1 kido
        kowasu;
    shushi


    visited[x, y, z] = visited[x, y, z] + 1;

    seisu min_val = 999;

    ronri go_right   = uso;
    ronri go_left    = uso;
    ronri go_down    = uso;
    ronri go_up      = uso;
    ronri go_forward = uso;
    ronri go_back    = uso;


    ronri path_to_exit = uso;


    sorenara { >_0 } < 0 kido go_right = shinri; path_to_exit = shinri; shushi
    sorenara { <_0 } < 0 kido go_left = shinri; path_to_exit = shinri; shushi
    sorenara { v_0 } < 0 kido go_down = shinri; path_to_exit = shinri; shushi
    sorenara { ^_0 } < 0 kido go_up = shinri; path_to_exit = shinri; shushi
    sorenara { o_0 } < 0 kido go_forward = shinri; path_to_exit = shinri; shushi
    sorenara { ~_0 } < 0 kido go_back = shinri; path_to_exit = shinri; shushi


    sorenara ~path_to_exit kido

        sorenara { >_0 } > 1 kido
            seisu val = visited[x + 1, y, z];
            sorenara val < min_val kido
                min_val = val; go_right = shinri;
                go_left = uso; go_down = uso; go_up = uso; go_forward = uso; go_back = uso;
            shushi
        shushi

        sorenara { <_0 } > 1 kido
            seisu val = visited[x - 1, y, z];
            sorenara val < min_val kido
                min_val = val; go_left = shinri;
                go_right = uso; go_down = uso; go_up = uso; go_forward = uso; go_back = uso;
            shushi
        shushi

        sorenara { o_0 } > 1 kido
            seisu val = visited[x, y - 1, z];
            sorenara val < min_val kido
                min_val = val; go_forward = shinri;
                go_right = uso; go_left = uso; go_down = uso; go_up = uso; go_back = uso;
            shushi
        shushi

        sorenara { ~_0 } > 1 kido
            seisu val = visited[x, y + 1, z];
            sorenara val < min_val kido
                min_val = val; go_back = shinri;
                go_right = uso; go_left = uso; go_down = uso; go_up = uso; go_forward = uso;
            shushi
        shushi

        sorenara { v_0 } > 1 kido
            seisu val = visited[x, y, z - 1];
            sorenara val < min_val kido
                min_val = val; go_down = shinri;
                go_right = uso; go_left = uso; go_up = uso; go_forward = uso; go_back = uso;
            shushi
        shushi

        sorenara { ^_0 } > 1 kido
            seisu val = visited[x, y, z + 1];
            sorenara val < min_val kido
                min_val = val; go_up = shinri;
                go_right = uso; go_left = uso; go_down = uso; go_forward = uso; go_back = uso;
            shushi
        shushi
    shushi


    sorenara go_right   kido { >_> }; shushi
    sorenara go_left    kido { <_< }; shushi
    sorenara go_forward kido { o_o }; shushi
    sorenara go_back    kido { ~_~ }; shushi
    sorenara go_up      kido { ^_^ }; shushi
    sorenara go_down    kido { v_v }; shushi

    x = { *_* } => x;
    y = { *_* } => y;
    z = { *_* } => z;

    step = step + 1;
shushi
